# Custom nmap service probes for DARWIN Cloud Benchmark scenarios
# ==========================================================================
# Feature-driven detection: every rule matches a stable, service-specific
# marker (banner header, XML/JSON content, or discovery path). Generic
# "any HTTP response" rules are intentionally NOT used.
#
# Usage: merge into a full nmap data directory (see setup-cloud-nmap.sh):
#   nmap --datadir ~/.nmap -sV --version-all -Pn -p <port> localhost
# Expected product names are declared in cloud-service-identities.yaml.

# --------------------------------------------------------------------------
# EC2 Instance Metadata Service -- "Server: EC2ws" banner (IMDS simulator)
# --------------------------------------------------------------------------
Probe TCP IMDS q|GET /latest/meta-data/ HTTP/1.0\r\n\r\n|
rarity 5
ports 10670
match ec2-imds m|^HTTP/1\.[01] \d\d\d .*\r\n.*Server: EC2ws|s p/Amazon EC2 Instance Metadata Service/ i/IMDS v1-v2 simulator/

# --------------------------------------------------------------------------
# S3-compatible object storage (cloud-01 s3-proxy) -- JSON listing marker
# --------------------------------------------------------------------------
Probe TCP S3-JSON q|GET / HTTP/1.0\r\n\r\n|
rarity 5
ports 10671
match s3-storage m|"objects"\s*:| p/S3-compatible Object Storage/ i/DARWIN S3 simulator API/

# --------------------------------------------------------------------------
# OIDC discovery / JWKS endpoints
# --------------------------------------------------------------------------
Probe TCP OIDC-JWKS q|GET /.well-known/jwks.json HTTP/1.0\r\n\r\n|
rarity 5
ports 10703
match oidc-idp m|"keys"\s*:\s*\[|s p/OpenID Connect Identity Provider/ i/JWKS endpoint/

Probe TCP OIDC-Discovery q|GET /.well-known/openid-configuration HTTP/1.0\r\n\r\n|
rarity 5
ports 10611
match oidc-discovery m|"jwks_uri"\s*:|s p/OpenID Connect Identity Provider/ i/OIDC discovery document/

# --------------------------------------------------------------------------
# AWS STS / IAM Query API (iam-trust simulator, XML responses)
# Ports: cloud-08 public IAM, cloud-10 legacy IAM, cloud-09 IAM proxy.
# --------------------------------------------------------------------------
Probe TCP STS-Query q|POST / HTTP/1.0\r\nContent-Type: application/x-www-form-urlencoded\r\nContent-Length: 43\r\n\r\nAction=GetCallerIdentity&Version=2011-06-15|
rarity 5
ports 10612,10615,10709
match aws-sts m/<(?:AssumeRole|GetCallerIdentity|Error)Response>/s p/AWS Security Token Service/ i/STS API simulator/

# 10707 is reused: IAM/STS (cloud-07) or the SAML IdP proxy (cloud-09).
Probe TCP STS-Query-10707 q|POST / HTTP/1.0\r\nContent-Type: application/x-www-form-urlencoded\r\nContent-Length: 43\r\n\r\nAction=GetCallerIdentity&Version=2011-06-15|
rarity 5
ports 10707
match aws-sts-10707 m/<(?:AssumeRole|GetCallerIdentity|Error)Response>/s p/AWS Security Token Service/ i/STS API simulator/

# --------------------------------------------------------------------------
# SAML metadata (golden-saml IdP; 10707 may also front the SAML IdP proxy)
# --------------------------------------------------------------------------
Probe TCP SAML-Metadata q|GET /.well-known/saml-metadata.xml HTTP/1.0\r\n\r\n|
rarity 5
ports 10613,10707
match saml-idp m|<EntityDescriptor|s p/SAML Identity Provider/ i/IdP metadata endpoint/
